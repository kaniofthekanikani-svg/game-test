rasutoday = None

def daycheck():
    global rasutoday, deiri

    now = datetime.now(ZoneInfo("Asia/Tokyo"))

    if rasutoday != now.date():
        deiri = []
        rasutoday = now.date()

    return now.hour >= okiro


# ===== 起動時チェック =====
@client.event

async def on_ready():
    print("bot起動")
    
    # 許可されてないサーバーから抜ける
    for guild in client.guilds:
        if str(guild.id) not in ALLOWED_GUILDS:
            await guild.leave()
            print(f"leave : {guild.id} : {guild.name}")
        else:
            print(f"allow : {guild.id} : {guild.name}")

# ===== 新規参加時チェック =====
@client.event
async def on_guild_join(guild):
    if str(guild.id) not in ALLOWED_GUILDS:
        await guild.leave()
        print(f"leave : {guild.id} : {guild.name}")
    else:
        print(f"allow : {guild.id} : {guild.name}")

@client.event
async def on_message(message):
    if message.guild is None:
        print('DM')
        return

    # サーバー制限
    if str(message.guild.id) not in ALLOWED_GUILDS:
        print('鯖制限')
        return
    if str(message.channel.id) not in TARGET_CHANNEL_ID:
        return
    
    
    msg = message.content
    
    
    if message.content == "" and message.stickers:
        if message.content == "" and message.stickers:
            match_found = False
            for sticker_item in message.stickers:
                if message.guild:
                    try:
                        full_sticker = await message.guild.fetch_sticker(sticker_item.id)

                        if full_sticker.description and tagsticker in full_sticker.description:
                            match_found = True
                            break
                    except Exception:
                        pass
                
        if match_found:
            if daycheck():
                if message.author.id not in deiri:
                    deiri.append(message.author.id)

                    count = len(deiri)
                    ttext = text.get(str(count), sonota)
                    
                    tttext = f"{ttext}\n-# {count}人目"
                    print(tttext)
                    await message.reply(tttext)
